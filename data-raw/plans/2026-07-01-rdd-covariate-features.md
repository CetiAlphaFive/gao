# RDD Covariate Features Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add report-composition and requester-party covariates to the `gao`
package — issuing division, product type, temporal/scope features, and requester
party/majority/chamber/bipartisan status — plus repair the noisy requester
committee/member fields, all wired into the accessor and the daily auto-update.

**Architecture:** Tier-1 features and the requester-party features are **derived
on the fly** inside `gao_links()`, exactly like the existing 82 one-hot indicator
columns (`.expand_indicators()`). No change to the shipped RDS schema or the
scraper is needed for derived columns — they recompute automatically for new
reports because they are deterministic functions of columns the scraper already
produces (`report_id`, `published`, `released`, `subject_terms`, `topics`,
`requester_members`, `requester_committees`). Requester party/majority uses a
**bundled VoteView-derived crosswalk** stored as internal package data
(`R/sysdata.rda`), so the join is offline and CRAN-safe. The requester-field
parse bugs are fixed at two levels: (a) the parse functions in the scraping layer
gain a defensive string-cleaner so all future scrapes and the daily gap-fill emit
clean values, and (b) a one-time `data-raw/` script string-cleans the existing
56k historical rows.

**Scoping decision (deliberate):** the package ships *neutral* temporal features
(`pub_month`, `pub_dow`, `fiscal_quarter`, `pub_fiscal_year`, `election_year`,
`release_lag_days`). It does **not** bake in the 1994 running variable
(`days_to_cutoff`, `post`) — those are analysis parameters and stay in the RDD
analysis script (`data-raw/rdd/`), because `gao` is a general GAO data package
headed for CRAN, not a paper-specific tool.

**Tech Stack:** R (base + roxygen2 + testthat 3e), `withr` (test fixtures),
`utils::download.file` (crosswalk build only, in `data-raw/`), VoteView
`HSall_members.csv`. No new package `Imports`.

```r
# the agency dropped division codes for the flat "GAO-YY-NNNN" scheme).
.gao_division_map <- c(
  NSIAD = "National Security & International Affairs",
  RCED  = "Resources, Community & Economic Development",
  GGD   = "General Government",
  HEHS  = "Health, Education & Human Services",
  HRD   = "Human Resources",
  AIMD  = "Accounting & Information Management",
  AFMD  = "Accounting & Financial Management",
  IMTEC = "Information Management & Technology",
  PEMD  = "Program Evaluation & Methodology",
  OGC   = "Office of General Counsel",
  OSI   = "Office of Special Investigations",
  OCG   = "Office of the Comptroller General",
  AA    = "Accounting & Auditing (legacy)",
  FGMSD = "Financial & General Management Studies (legacy)",
  PLRD  = "Procurement, Logistics & Readiness (legacy)",
  IPE   = "Institute for Program Evaluation (legacy)",
  CED   = "Community & Economic Development (legacy)",
  ID    = "International Division (legacy)",
  FPCD  = "Federal Personnel & Compensation (legacy)",
  GAO   = "Post-2000 (division code dropped)"
)

#' Alpha division/office prefix from a report ID
#' @keywords internal
#' @noRd
.gao_prefix <- function(report_id) {
  id <- toupper(trimws(as.character(report_id)))
  id <- sub("^T-", "", id)          # testimony carries the division after "T-"
  id <- sub("^B-.*$", "B", id)      # legal decisions collapse to bare "B"
  pre <- sub("^([A-Z]+).*$", "\\1", id)
  pre[grepl("^[0-9]", id)] <- NA_character_   # pure-numeric legacy IDs: unknown
  pre[!nzchar(pre)] <- NA_character_
  pre
}

#' Issuing GAO division/office for each report
#' @keywords internal
#' @noRd
.gao_issuing_division <- function(report_id) {
  pre <- .gao_prefix(report_id)
  div <- unname(.gao_division_map[pre])
  div[!is.na(pre) & pre == "B"]           <- "Legal decision (Comptroller General)"
  div[is.na(div) & !is.na(pre)]           <- "Other/uncommon office"
  div[is.na(pre)]                         <- NA_character_
  div
}

#' Product type: report / testimony / correspondence / legal_decision / legal_other
#' @keywords internal
#' @noRd
.gao_product_type <- function(report_id) {
  id <- toupper(trimws(as.character(report_id)))
  pt <- rep("report", length(id))
  pt[grepl("^T-", id) | grepl("T$", id)] <- "testimony"
  pt[grepl("^B-", id)]                    <- "legal_decision"
  pt[grepl("[0-9]R$", id)]                <- "correspondence"
  pt[grepl("^OGC", id) & pt == "report"]  <- "legal_other"
  pt[is.na(id) | !nzchar(id)]             <- NA_character_
  pt
}
```

```r
# append to R/features.R

#' Count semicolon-delimited items (0 for NA/empty)
#' @keywords internal
#' @noRd
.count_delimited <- function(x) {
  ifelse(is.na(x) | !nzchar(x), 0L, lengths(strsplit(x, ";")))
}

#' Add all Tier-1 derived columns to a report data.frame
#'
#' Deterministic functions of columns the scraper already produces. Called by
#' [gao_links()]. Guarded there so it runs once per load.
#' @keywords internal
#' @noRd
.expand_features <- function(df) {
  pub <- as.Date(df$published)
  rel <- as.Date(df$released)
  mo  <- as.integer(format(pub, "%m"))
  yr  <- as.integer(format(pub, "%Y"))

  df$issuing_division <- .gao_issuing_division(df$report_id)
  df$product_type     <- .gao_product_type(df$report_id)

  df$pub_month       <- mo
  df$pub_dow         <- as.integer(format(pub, "%u"))                # 1=Mon..7=Sun
  df$pub_fiscal_year <- ifelse(is.na(mo), NA_integer_,
                               ifelse(mo >= 10L, yr + 1L, yr))
  df$fiscal_quarter  <- ((mo + 2L) %% 12L) %/% 3L + 1L              # FY starts Oct
  df$election_year   <- ifelse(is.na(yr), NA_integer_,
                               as.integer(yr %% 2L == 0L))
  df$release_lag_days <- as.integer(rel - pub)

  df$n_topics        <- .count_delimited(df$topics)
  df$n_subject_terms <- .count_delimited(df$subject_terms)
  df
}
```

```r
# data-raw/build_member_crosswalk.R
# Builds internal crosswalk data shipped as R/sysdata.rda.
# Run manually: Rscript data-raw/build_member_crosswalk.R
# Source: VoteView (Lewis et al.), public domain CSV.

url <- "https://voteview.com/static/data/out/members/HSall_members.csv"
raw <- read.csv(url, stringsAsFactors = FALSE)

# Keep House (chamber "House") and Senate ("Senate"); drop President rows.
raw <- raw[raw$chamber %in% c("House", "Senate"), ]

party_label <- function(code) {
  ifelse(code == 100, "D", ifelse(code == 200, "R", "Other"))
}

# --- .member_party: one row per (congress, chamber, member) ---
.member_party <- data.frame(
  congress   = as.integer(raw$congress),
  chamber    = ifelse(raw$chamber == "House", "House", "Senate"),
  bioname    = raw$bioname,                       # "LUGAR, Richard Green"
  state      = raw$state_abbrev,
  party      = party_label(raw$party_code),
  stringsAsFactors = FALSE
)
# Normalized match key: LAST + first initial (built with the same normalizer the
# runtime uses — see R/requester_party.R::.normalize_member_name).
# The build sources that file so the key logic stays single-definition.
source("R/requester_party.R", local = TRUE)
.member_party$match_key <- .normalize_member_name(
  paste(sub("^([^,]+),.*$", "\\1", .member_party$bioname),   # last name
        sub("^[^,]+,\\s*(\\S+).*$", "\\1", .member_party$bioname)) # first token
)

# Drop within-(congress, chamber, match_key) collisions across parties: ambiguous
# keys are set to NA party so the runtime returns "unknown" rather than guessing.
agg <- stats::aggregate(party ~ congress + chamber + match_key,
                        data = .member_party,
                        FUN = function(p) length(unique(p)))
ambiguous <- agg[agg$party > 1, c("congress", "chamber", "match_key")]
.member_party <- merge(.member_party, cbind(ambiguous, ambig = TRUE),
                       all.x = TRUE)
.member_party$party[which(.member_party$ambig)] <- NA_character_
.member_party$ambig <- NULL
.member_party <- unique(.member_party[, c("congress", "chamber",
                                          "match_key", "party")])

# --- .majority_by_congress: majority party per (congress, chamber) ---
seats <- stats::aggregate(
  list(n = rep(1L, nrow(raw))),
  by = list(congress = as.integer(raw$congress),
            chamber = ifelse(raw$chamber == "House", "House", "Senate"),
            party = party_label(raw$party_code)),
  FUN = sum
)
.majority_by_congress <- do.call(rbind, by(seats, list(seats$congress, seats$chamber),
  function(g) {
    g <- g[g$party %in% c("D", "R"), ]
    data.frame(congress = g$congress[1], chamber = g$chamber[1],
               majority_party = g$party[which.max(g$n)],
               stringsAsFactors = FALSE)
  }))
rownames(.majority_by_congress) <- NULL

usethis::use_data(.member_party, .majority_by_congress,
                  internal = TRUE, overwrite = TRUE)
cat("Wrote R/sysdata.rda:",
    nrow(.member_party), "member rows,",
    nrow(.majority_by_congress), "majority rows\n")
```

```r
# R/requester_party.R

#' Congress number active on a given date
#'
#' The Nth Congress convenes Jan 3 of the odd year `1789 + 2*(N-1)`.
#' @param date A Date (or vector).
#' @keywords internal
#' @noRd
.congress_for_date <- function(date) {
  date <- as.Date(date)
  yr <- as.integer(format(date, "%Y"))
  # Base congress if the year were the convening (odd) year.
  cong <- (yr - 1789L) %/% 2L + 1L
  # In an odd year before Jan 3, the previous Congress still sits.
  before.swear <- (yr %% 2L == 1L) & (date < as.Date(paste0(yr, "-01-03")))
  cong[before.swear] <- cong[before.swear] - 1L
  as.integer(cong)
}

#' Normalize a member name to a "lastname firstinitial" match key
#'
#' Strips honorifics, nicknames in parentheses, middle initials, punctuation.
#' Handles "First M. Last" and "Last, First" inputs.
#' @keywords internal
#' @noRd
.normalize_member_name <- function(name) {
  vapply(name, function(nm) {
    if (is.na(nm) || !nzchar(trimws(nm))) return(NA_character_)
    x <- nm
    x <- gsub("\\([^)]*\\)", " ", x)                 # drop "(Sonny)" nicknames
    x <- gsub("[^A-Za-z ,]", " ", x)                 # keep letters/comma/space
    x <- trimws(gsub("\\s+", " ", x))
    if (grepl(",", x)) {                             # "Last, First"
      parts <- strsplit(x, ",")[[1]]
      last  <- trimws(parts[1])
      first <- trimws(parts[2])
    } else {                                         # "First M. Last"
      toks  <- strsplit(x, " ")[[1]]
      toks  <- toks[nchar(toks) > 0]
      last  <- toks[length(toks)]
      first <- toks[1]
    }
    tolower(paste(last, substr(first, 1, 1)))
  }, character(1), USE.NAMES = FALSE)
}

#' Requester party / chamber / majority / bipartisan for one report
#'
#' @param members Character scalar: semicolon-delimited requester member string.
#' @param committees Character scalar: semicolon-delimited committee string
#'   (used only to infer chamber when member names are absent).
#' @param published Character/Date scalar publication date.
#' @return A one-row list with requester_party ("R"/"D"/"mixed"/NA),
#'   requester_majority_status ("majority"/"minority"/"mixed"/NA),
#'   requester_chamber ("House"/"Senate"/"both"/NA), requester_bipartisan (logical).
#' @keywords internal
#' @noRd
.requester_party_features <- function(members, committees, published) {
  na.out <- list(requester_party = NA_character_,
                 requester_majority_status = NA_character_,
                 requester_chamber = NA_character_,
                 requester_bipartisan = NA)

  cong <- .congress_for_date(published)
  if (is.na(cong)) return(na.out)

  # --- chamber from committee annotations "(Senate)"/"(House)" ---
  chamber <- NA_character_
  if (!is.na(committees) && nzchar(committees)) {
    has.sen <- grepl("Senate", committees)
    has.hou <- grepl("House",  committees)
    chamber <- if (has.sen && has.hou) "both"
               else if (has.sen) "Senate"
               else if (has.hou) "House" else NA_character_
  }

  if (is.na(members) || !nzchar(members)) {
    return(list(requester_party = NA_character_,
                requester_majority_status = NA_character_,
                requester_chamber = chamber,
                requester_bipartisan = NA))
  }

  member.strs <- trimws(strsplit(members, ";")[[1]])
  keys <- .normalize_member_name(member.strs)

  # look up party for the report's Congress, trying Senate then House if the
  # member's chamber is unknown
  cw <- .member_party[.member_party$congress == cong, ]
  parties <- vapply(keys, function(k) {
    if (is.na(k)) return(NA_character_)
    hit <- cw$party[cw$match_key == k]
    hit <- hit[!is.na(hit)]
    if (length(hit) >= 1L) hit[1] else NA_character_
  }, character(1), USE.NAMES = FALSE)

  parties <- parties[!is.na(parties)]
  if (length(parties) == 0L) {
    return(list(requester_party = NA_character_,
                requester_majority_status = NA_character_,
                requester_chamber = chamber,
                requester_bipartisan = NA))
  }

  uniq <- unique(parties)
  req.party <- if (length(uniq) == 1L) uniq else "mixed"
  bipartisan <- ("R" %in% parties) && ("D" %in% parties)

  # majority status: compare each party to the chamber majority for the Congress
  maj.chamber <- if (!is.na(chamber) && chamber %in% c("House", "Senate")) chamber else "House"
  maj <- .majority_by_congress$majority_party[
    .majority_by_congress$congress == cong &
    .majority_by_congress$chamber == maj.chamber]
  maj <- if (length(maj) == 1L) maj else NA_character_

  maj.status <- if (is.na(maj)) NA_character_
    else {
      in.maj <- parties == maj
      if (all(in.maj)) "majority"
      else if (!any(in.maj)) "minority"
      else "mixed"
    }

  list(requester_party = req.party,
       requester_majority_status = maj.status,
       requester_chamber = chamber,
       requester_bipartisan = bipartisan)
}
```

```r
# R/requester_clean.R

#' Clean parsed requester committee/member strings
#'
#' Defensive post-parse cleanup for known noise patterns produced by the
#' upstream HTML/PDF parsers: an injected "GAO" token inside committee names,
#' "RELEASED" bleed from "Publicly Released" date text, and OCR mid-word caps.
#' Applied on the return paths of the requester parsers and by the one-time
#' historical cleanup in data-raw/clean_requester_fields.R.
#'
#' @param x Character vector of semicolon-delimited requester strings.
#' @return Cleaned character vector, same length; NA in -> NA out.
#' @keywords internal
#' @noRd
.clean_requester_string <- function(x) {
  clean1 <- function(s) {
    if (is.na(s)) return(NA_character_)
    if (!nzchar(s)) return(s)
    items <- trimws(strsplit(s, ";")[[1]])
    items <- vapply(items, function(it) {
      # 1. injected "GAO" after "Committee on"/"Subcommittee on"
      it <- sub("((?:Sub)?Committee on) GAO ", "\\1 ", it, perl = TRUE)
      # 2. drop "RELEASED" / "Publicly Released" noise tokens
      it <- gsub("\\s*(?:Publicly )?RELEASED\\b", "", it, perl = TRUE)
      # 3. fix OCR mid-word caps like "LUgar" -> "Lugar" (upper then lower run)
      it <- gsub("\\b([A-Z])([A-Z])([a-z]+)", "\\1\\L\\2\\3", it, perl = TRUE)
      trimws(gsub("\\s+", " ", it))
    }, character(1), USE.NAMES = FALSE)
    items <- items[nzchar(items)]
    if (length(items) == 0L) return(NA_character_)
    paste(items, collapse = "; ")
  }
  vapply(x, clean1, character(1), USE.NAMES = FALSE)
}
```
